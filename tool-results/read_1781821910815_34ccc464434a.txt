   214→   214→export const alertsRepository = {
   215→   215→  async findMany(filters: {
   216→   216→    status?: string;
   217→   217→    priority?: string;
   218→   218→    type?: string;
   219→   219→  } = {}): Promise<AlertRecord[]> {
   220→   220→    if (isSupabaseMode()) {
   221→   221→      const supabase = getSupabaseServer();
   222→   222→      if (supabase) {
   223→   223→        let query = supabase
   224→   224→          .from('alerts')
   225→   225→          .select(
   226→   226→            '*, commissariat:commissariats(id,name,code), assignedTo:officers(id,first_name,last_name,rank,matricule)'
   227→   227→          )
   228→   228→          .order('created_at', { ascending: false });
   229→   229→        if (filters.status) query = query.eq('status', filters.status);
   230→   230→        if (filters.priority) query = query.eq('priority', filters.priority);
   231→   231→        if (filters.type) query = query.eq('type', filters.type);
   232→   232→        const { data, error } = await query;
   233→   233→        if (error) throw error;
   234→   234→        return (data || []).map((a: any) => ({
   235→   235→          id: a.id,
   236→   236→          reference: a.reference,
   237→   237→          type: a.type,
   238→   238→          priority: a.priority,
   239→   239→          status: a.status,
   240→   240→          description: a.description,
   241→   241→          location: a.location,
   242→   242→          latitude: a.latitude,
   243→   243→          longitude: a.longitude,
   244→   244→          citizenName: a.citizen_name,
   245→   245→          citizenPhone: a.citizen_phone,
   246→   246→          citizenId: a.citizen_id,
   247→   247→          commissariatId: a.commissariat_id,
   248→   248→          assignedToId: a.assigned_to_id,
   249→   249→          responseTime: a.response_time,
   250→   250→          caseId: a.case_id,
   251→   251→          createdAt: a.created_at,
   252→   252→          updatedAt: a.updated_at,
   253→   253→          commissariat: a.commissariat,
   254→   254→          assignedTo: a.assignedTo,
   255→   255→        }));
   256→   256→      }
   257→   257→    }
   258→   258→    // Mode Prisma
   259→   259→    const where: Record<string, string> = {};
   260→   260→    if (filters.status) where.status = filters.status;
   261→   261→    if (filters.priority) where.priority = filters.priority;
   262→   262→    if (filters.type) where.type = filters.type;
   263→   263→    const alerts = await db.alert.findMany({
   264→   264→      where,
   265→   265→      orderBy: { createdAt: 'desc' },
   266→   266→      include: {
   267→   267→        commissariat: { select: { id: true, name: true, code: true } },
   268→   268→        assignedTo: {
   269→   269→          select: { id: true, firstName: true, lastName: true, rank: true, matricule: true },
   270→   270→        },
   271→   271→      },
   272→   272→    });
   273→   273→    return alerts.map(mapAlertFromPrisma);
   274→   274→  },

   276→   276→  async create(input: {
   277→   277→    type: string;
   278→   278→    priority: string;
   279→   279→    description: string;
   280→   280→    location: string;
   281→   281→    latitude?: number | null;
   282→   282→    longitude?: number | null;
   283→   283→    citizenName?: string | null;
   284→   284→    citizenPhone?: string | null;
   285→   285→    citizenId?: string | null;
   286→   286→    commissariatId: string;
   287→   287→    assignedToId?: string | null;
   288→   288→  }): Promise<AlertRecord> {
   289→   289→    // Génération de la référence
   290→   290→    const year = new Date().getFullYear();
   291→   291→    const prefix = `ALT-${year}-`;
   292→   292→
   293→   293→    if (isSupabaseMode()) {
   294→   294→      const supabase = getSupabaseServer();
   295→   295→      if (supabase) {
   296→   296→        const { data: last } = await supabase
   297→   297→          .from('alerts')
   298→   298→          .select('reference')
   299→   299→          .like('reference', `${prefix}%`)
   300→   300→          .order('reference', { ascending: false })
   301→   301→          .limit(1);
   302→   302→        const nextNum = last && last[0] ? parseInt(last[0].reference.split('-')[2], 10) + 1 : 1;
   303→   303→        const reference = `${prefix}${String(nextNum).padStart(3, '0')}`;
   304→   304→
   305→   305→        const { data, error } = await supabase
   306→   306→          .from('alerts')
   307→   307→          .insert({
   308→   308→            reference,
   309→   309→            type: input.type,
   310→   310→            priority: input.priority,
   311→   311→            status: 'recue',
   312→   312→            description: input.description,
   313→   313→            location: input.location,
   314→   314→            latitude: input.latitude ?? null,
   315→   315→            longitude: input.longitude ?? null,
   316→   316→            citizen_name: input.citizenName ?? null,
   317→   317→            citizen_phone: input.citizenPhone ?? null,
   318→   318→            citizen_id: input.citizenId ?? null,
   319→   319→            commissariat_id: input.commissariatId,
   320→   320→            assigned_to_id: input.assignedToId ?? null,
   321→   321→          })
   322→   322→          .select(
   323→   323→            '*, commissariat:commissariats(id,name,code), assignedTo:officers(id,first_name,last_name,rank,matricule)'
   324→   324→          )
   325→   325→          .single();
   326→   326→        if (error) throw error;
   327→   327→        const a: any = data;
   328→   328→        return {
   329→   329→          id: a.id,
   330→   330→          reference: a.reference,
   331→   331→          type: a.type,
   332→   332→          priority: a.priority,
   333→   333→          status: a.status,
   334→   334→          description: a.description,
   335→   335→          location: a.location,
   336→   336→          latitude: a.latitude,
   337→   337→          longitude: a.longitude,
   338→   338→          citizenName: a.citizen_name,
   339→   339→          citizenPhone: a.citizen_phone,
   340→   340→          citizenId: a.citizen_id,
   341→   341→          commissariatId: a.commissariat_id,
   342→   342→          assignedToId: a.assigned_to_id,
   343→   343→          responseTime: a.response_time,
   344→   344→          caseId: a.case_id,
   345→   345→          createdAt: a.created_at,
   346→   346→          updatedAt: a.updated_at,
   347→   347→          commissariat: a.commissariat,
   348→   348→          assignedTo: a.assignedTo,
   349→   349→        };
   350→   350→      }
   351→   351→    }
   352→   352→
   353→   353→    // Mode Prisma
   354→   354→    const lastAlert = await db.alert.findFirst({
   355→   355→      where: { reference: { startsWith: prefix } },
   356→   356→      orderBy: { reference: 'desc' },
   357→   357→    });
   358→   358→    const nextNum = lastAlert ? parseInt(lastAlert.reference.split('-')[2], 10) + 1 : 1;
   359→   359→    const reference = `${prefix}${String(nextNum).padStart(3, '0')}`;
   360→   360→
   361→   361→    const alert = await db.alert.create({
   362→   362→      data: {
   363→   363→        reference,
   364→   364→        type: input.type,
   365→   365→        priority: input.priority,
   366→   366→        status: 'recue',
   367→   367→        description: input.description,
   368→   368→        location: input.location,
   369→   369→        latitude: input.latitude ?? null,
   370→   370→        longitude: input.longitude ?? null,
   371→   371→        citizenName: input.citizenName ?? null,
   372→   372→        citizenPhone: input.citizenPhone ?? null,
   373→   373→        citizenId: input.citizenId ?? null,
   374→   374→        commissariatId: input.commissariatId,
   375→   375→        assignedToId: input.assignedToId ?? null,
   376→   376→      },
   377→   377→      include: {
   378→   378→        commissariat: { select: { id: true, name: true, code: true } },
   379→   379→        assignedTo: {
   380→   380→          select: { id: true, firstName: true, lastName: true, rank: true, matricule: true },
   381→   381→        },
   382→   382→      },
   383→   383→    });
   384→   384→
   385→   385→    // Incrémenter le compteur d'alertes du citoyen
   386→   386→    if (input.citizenId) {
   387→   387→      await db.citizen.update({
   388→   388→        where: { id: input.citizenId },
   389→   389→        data: { totalAlerts: { increment: 1 } },
   390→   390→      }).catch(() => {});
   391→   391→    }
   392→   392→
   393→   393→    return mapAlertFromPrisma(alert);
   394→   394→  },
