   214→export const alertsRepository = {
   215→  async findMany(filters: {
   216→    status?: string;
   217→    priority?: string;
   218→    type?: string;
   219→  } = {}): Promise<AlertRecord[]> {
   220→    if (isSupabaseMode()) {
   221→      const supabase = getSupabaseServer();
   222→      if (supabase) {
   223→        let query = supabase
   224→          .from('alerts')
   225→          .select(
   226→            '*, commissariat:commissariats(id,name,code), assignedTo:officers(id,first_name,last_name,rank,matricule)'
   227→          )
   228→          .order('created_at', { ascending: false });
   229→        if (filters.status) query = query.eq('status', filters.status);
   230→        if (filters.priority) query = query.eq('priority', filters.priority);
   231→        if (filters.type) query = query.eq('type', filters.type);
   232→        const { data, error } = await query;
   233→        if (error) throw error;
   234→        return (data || []).map((a: any) => ({
   235→          id: a.id,
   236→          reference: a.reference,
   237→          type: a.type,
   238→          priority: a.priority,
   239→          status: a.status,
   240→          description: a.description,
   241→          location: a.location,
   242→          latitude: a.latitude,
   243→          longitude: a.longitude,
   244→          citizenName: a.citizen_name,
   245→          citizenPhone: a.citizen_phone,
   246→          citizenId: a.citizen_id,
   247→          commissariatId: a.commissariat_id,
   248→          assignedToId: a.assigned_to_id,
   249→          responseTime: a.response_time,
   250→          caseId: a.case_id,
   251→          createdAt: a.created_at,
   252→          updatedAt: a.updated_at,
   253→          commissariat: a.commissariat,
   254→          assignedTo: a.assignedTo,
   255→        }));
   256→      }
   257→    }
   258→    // Mode Prisma
   259→    const where: Record<string, string> = {};
   260→    if (filters.status) where.status = filters.status;
   261→    if (filters.priority) where.priority = filters.priority;
   262→    if (filters.type) where.type = filters.type;
   263→    const alerts = await db.alert.findMany({
   264→      where,
   265→      orderBy: { createdAt: 'desc' },
   266→      include: {
   267→        commissariat: { select: { id: true, name: true, code: true } },
   268→        assignedTo: {
   269→          select: { id: true, firstName: true, lastName: true, rank: true, matricule: true },
   270→        },
   271→      },
   272→    });
   273→    return alerts.map(mapAlertFromPrisma);
   274→  },

   276→  async create(input: {
   277→    type: string;
   278→    priority: string;
   279→    description: string;
   280→    location: string;
   281→    latitude?: number | null;
   282→    longitude?: number | null;
   283→    citizenName?: string | null;
   284→    citizenPhone?: string | null;
   285→    citizenId?: string | null;
   286→    commissariatId: string;
   287→    assignedToId?: string | null;
   288→  }): Promise<AlertRecord> {
   289→    // Génération de la référence
   290→    const year = new Date().getFullYear();
   291→    const prefix = `ALT-${year}-`;
   292→
   293→    if (isSupabaseMode()) {
   294→      const supabase = getSupabaseServer();
   295→      if (supabase) {
   296→        const { data: last } = await supabase
   297→          .from('alerts')
   298→          .select('reference')
   299→          .like('reference', `${prefix}%`)
   300→          .order('reference', { ascending: false })
   301→          .limit(1);
   302→        const nextNum = last && last[0] ? parseInt(last[0].reference.split('-')[2], 10) + 1 : 1;
   303→        const reference = `${prefix}${String(nextNum).padStart(3, '0')}`;
   304→
   305→        const { data, error } = await supabase
   306→          .from('alerts')
   307→          .insert({
   308→            reference,
   309→            type: input.type,
   310→            priority: input.priority,
   311→            status: 'recue',
   312→            description: input.description,
   313→            location: input.location,
   314→            latitude: input.latitude ?? null,
   315→            longitude: input.longitude ?? null,
   316→            citizen_name: input.citizenName ?? null,
   317→            citizen_phone: input.citizenPhone ?? null,
   318→            citizen_id: input.citizenId ?? null,
   319→            commissariat_id: input.commissariatId,
   320→            assigned_to_id: input.assignedToId ?? null,
   321→          })
   322→          .select(
   323→            '*, commissariat:commissariats(id,name,code), assignedTo:officers(id,first_name,last_name,rank,matricule)'
   324→          )
   325→          .single();
   326→        if (error) throw error;
   327→        const a: any = data;
   328→        return {
   329→          id: a.id,
   330→          reference: a.reference,
   331→          type: a.type,
   332→          priority: a.priority,
   333→          status: a.status,
   334→          description: a.description,
   335→          location: a.location,
   336→          latitude: a.latitude,
   337→          longitude: a.longitude,
   338→          citizenName: a.citizen_name,
   339→          citizenPhone: a.citizen_phone,
   340→          citizenId: a.citizen_id,
   341→          commissariatId: a.commissariat_id,
   342→          assignedToId: a.assigned_to_id,
   343→          responseTime: a.response_time,
   344→          caseId: a.case_id,
   345→          createdAt: a.created_at,
   346→          updatedAt: a.updated_at,
   347→          commissariat: a.commissariat,
   348→          assignedTo: a.assignedTo,
   349→        };
   350→      }
   351→    }
   352→
   353→    // Mode Prisma
   354→    const lastAlert = await db.alert.findFirst({
   355→      where: { reference: { startsWith: prefix } },
   356→      orderBy: { reference: 'desc' },
   357→    });
   358→    const nextNum = lastAlert ? parseInt(lastAlert.reference.split('-')[2], 10) + 1 : 1;
   359→    const reference = `${prefix}${String(nextNum).padStart(3, '0')}`;
   360→
   361→    const alert = await db.alert.create({
   362→      data: {
   363→        reference,
   364→        type: input.type,
   365→        priority: input.priority,
   366→        status: 'recue',
   367→        description: input.description,
   368→        location: input.location,
   369→        latitude: input.latitude ?? null,
   370→        longitude: input.longitude ?? null,
   371→        citizenName: input.citizenName ?? null,
   372→        citizenPhone: input.citizenPhone ?? null,
   373→        citizenId: input.citizenId ?? null,
   374→        commissariatId: input.commissariatId,
   375→        assignedToId: input.assignedToId ?? null,
   376→      },
   377→      include: {
   378→        commissariat: { select: { id: true, name: true, code: true } },
   379→        assignedTo: {
   380→          select: { id: true, firstName: true, lastName: true, rank: true, matricule: true },
   381→        },
   382→      },
   383→    });
   384→
   385→    // Incrémenter le compteur d'alertes du citoyen
   386→    if (input.citizenId) {
   387→      await db.citizen.update({
   388→        where: { id: input.citizenId },
   389→        data: { totalAlerts: { increment: 1 } },
   390→      }).catch(() => {});
   391→    }
   392→
   393→    return mapAlertFromPrisma(alert);
   394→  },

   396→  async updateStatus(
   397→    id: string,
   398→    status: string,
   399→    extra?: { assignedToId?: string | null; responseTime?: number | null }
   400→  ): Promise<AlertRecord | null> {
   401→    if (isSupabaseMode()) {
   402→      const supabase = getSupabaseServer();
   403→      if (supabase) {
   404→        const update: Record<string, unknown> = { status };
   405→        if (extra?.assignedToId !== undefined) update.assigned_to_id = extra.assignedToId;
   406→        if (extra?.responseTime !== undefined) update.response_time = extra.responseTime;
   407→        const { data, error } = await supabase
   408→          .from('alerts')
   409→          .update(update)
   410→          .eq('id', id)
   411→          .select(
   412→            '*, commissariat:commissariats(id,name,code), assignedTo:officers(id,first_name,last_name,rank,matricule)'
   413→          )
   414→          .single();
   415→        if (error) throw error;
   416→        const a: any = data;
   417→        return {
   418→          id: a.id,
   419→          reference: a.reference,
   420→          type: a.type,
   421→          priority: a.priority,
   422→          status: a.status,
   423→          description: a.description,
   424→          location: a.location,
   425→          latitude: a.latitude,
   426→          longitude: a.longitude,
   427→          citizenName: a.citizen_name,
   428→          citizenPhone: a.citizen_phone,
   429→          citizenId: a.citizen_id,
   430→          commissariatId: a.commissariat_id,
   431→          assignedToId: a.assigned_to_id,
   432→          responseTime: a.response_time,
   433→          caseId: a.case_id,
   434→          createdAt: a.created_at,
   435→          updatedAt: a.updated_at,
   436→          commissariat: a.commissariat,
   437→          assignedTo: a.assignedTo,
   438→        };
   439→      }
   440→    }
   441→    const alert = await db.alert.update({
   442→      where: { id },
   443→      data: {
   444→        status,
   445→        assignedToId: extra?.assignedToId,
   446→        responseTime: extra?.responseTime,
   447→      },
   448→      include: {
   449→        commissariat: { select: { id: true, name: true, code: true } },
   450→        assignedTo: {
   451→          select: { id: true, firstName: true, lastName: true, rank: true, matricule: true },
   452→        },
   453→      },
   454→    });
   455→    return mapAlertFromPrisma(alert);
   456→  },

   591→    const reference = `${prefix}${String(nextNum).padStart(3, '0')}`;
   592→
   593→    const complaint = await db.complaint.create({
   594→      data: {
   595→        reference,
   596→        type: input.type,
   597→        status: 'soumise',
   598→        description: input.description,
   599→        location: input.location ?? null,
   600→        plaintiffName: input.plaintiffName,
   601→        plaintiffPhone: input.plaintiffPhone,

   603→        plaintiffAddr: input.plaintiffAddr ?? null,
   604→        citizenId: input.citizenId ?? null,
   605→        commissariatId: input.commissariatId,
   606→      },

   609→        reviewedBy: { select: { id: true, firstName: true, lastName: true, rank: true } },
   610→      },
   611→    });

   613→    if (input.citizenId) {
   614→      await db.citizen.update({

   616→        data: { totalComplaints: { increment: 1 } },
   617→      }).catch(() => {});

   619→
   620→    return mapComplaintFromPrisma(complaint);
   621→  },

   786→    city?: string | null;
   787→    commune?: string | null;

   789→    commissariatId?: string | null;
   790→    passwordHash?: string | null;

   794→

   796→      const supabase = getSupabaseServer();
   797→      if (supabase) {

   799→          .from('citizens')
   800→          .select('reference')
   801→          .like('reference', `${prefix}%`)
   802→          .order('reference', { ascending: false })
   803→          .limit(1);
   804→        const nextNum = last && last[0] ? parseInt(last[0].reference.split('-')[2], 10) + 1 : 1;
   805→        const reference = `${prefix}${String(nextNum).padStart(3, '0')}`;
   806→

   808→          .from('citizens')

   812→            last_name: input.lastName,

   814→            email: input.email ?? null,
   815→            gender: input.gender ?? null,
   816→            city: input.city ?? null,
   817→            commune: input.commune ?? null,
   818→            address: input.address ?? null,
   819→            commissariat_id: input.commissariatId ?? null,
   820→            password_hash: input.passwordHash ?? null,
   821→            status: 'actif',
   822→            verified: false,
   823→          })
   824→          .select('*, commissariat:commissariats(id,name,code)')
   825→          .single();
   826→        if (error) throw error;
   827→        const c: any = data;

   830→          reference: c.reference,

   850→          updatedAt: c.updated_at,
   851→          commissariat: c.commissariat,
   852→        };
   853→      }
   854→    }

   873→        address: input.address ?? null,
   874→        commissariatId: input.commissariatId ?? null,
   875→        passwordHash: input.passwordHash ?? null,
   876→      },
   877→      include: { commissariat: { select: { id: true, name: true, code: true } } },
   878→    });
   879→    return mapCitizenFromPrisma(citizen);
   880→  },
